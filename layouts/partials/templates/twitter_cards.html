{{- /* Overrides PaperMod's templates/twitter_cards.html. Uses the same per-post OGP image as templates/opengraph.html (static/images/ogp/.png generated by tcardgen) so the Twitter/X card shows the right image instead of the missing site-wide default. Keeping the Twitter tags in one place also avoids emitting duplicate twitter:image meta tags. */ -}} {{- $ogImage := "" -}} {{- if and .IsPage .File -}} {{- $fileName := path.BaseName .File.Path -}} {{- $ogImagePath := printf "images/ogp/%s.png" (strings.TrimSuffix ".md" $fileName) -}} {{- if fileExists (printf "static/%s" $ogImagePath) -}} {{- $ogImage = printf "/%s" $ogImagePath -}} {{- end -}} {{- end -}} {{- if not $ogImage -}} {{- if .Params.images -}} {{- $ogImage = index .Params.images 0 -}} {{- else if .Site.Params.images -}} {{- $ogImage = index .Site.Params.images 0 -}} {{- end -}} {{- end -}} {{- if $ogImage -}} {{- else -}} {{- end -}}
